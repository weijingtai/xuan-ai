  1. 独立子模块 vs 集成开发

  方案对比：
  维度: 代码复用
  独立子模块 (ai_core): ✅ 所有技法共享同一套 AI 基础设施
  集成到各子模块: ❌ 每个模块各自实现，重复代码
  ────────────────────────────────────────
  维度: 版本管理
  独立子模块 (ai_core): ✅ 独立演进，不影响技法模块稳定性
  集成到各子模块: ❌ AI 更新可能破坏技法模块
  ────────────────────────────────────────
  维度: 依赖关系
  独立子模块 (ai_core): 技法模块 → ai_core (单向依赖)
  集成到各子模块: 耦合在一起
  ────────────────────────────────────────
  维度: Agent 互调
  独立子模块 (ai_core): ✅ 天然支持跨模块调用
  集成到各子模块: ❌ 需要额外抽象
  ────────────────────────────────────────
  维度: 团队协作
  独立子模块 (ai_core): ✅ AI 和技法可以并行开发
  集成到各子模块: ❌ 容易冲突

  2. Prompt 管理系统

  您说得对，当前设计过于粗浅。一个完善的 Prompt 管理系统应该包括：

  /// Prompt 模板实体
  class PromptTemplate {
    final String id;
    final String name;
    final String description;
    final String category;          // 分类: system/user/few_shot
    final String template;          // 模板内容，支持变量插值 {variable}
    final List<String> variables;   // 可用变量列表
    final String? parentId;         // 继承自哪个模板
    final Map<String, dynamic>? metadata;
    final int version;
    final DateTime createdAt;
    final DateTime updatedAt;
  }

  /// Prompt 管理器
  abstract class PromptManager {
    /// CRUD
    Future<PromptTemplate> create(PromptTemplate template);
    Future<PromptTemplate> update(String id, PromptTemplate template);
    Future<void> delete(String id);
    Future<PromptTemplate?> getById(String id);

    /// 查询
    Future<List<PromptTemplate>> listByCategory(String category);
    Future<List<PromptTemplate>> listBySkill(String skillId);

    /// 渲染
    String render(PromptTemplate template, Map<String, dynamic> variables);

    /// 版本管理
    Future<List<PromptTemplate>> getVersionHistory(String id);
    Future<PromptTemplate> rollback(String id, int version);
  }

  /// Agent 定义

  }

  /// Agent 调用请求
  class AgentInvocation {
    final String fromAgentId;      // 调用方 Agent
    final String toAgentId;        // 被调用 Agent
    final String purpose;          // 调用目的
    final Map<String, dynamic> context;  // 上下文数据
    final String? instruction;     // 具体指令
  }

  /// Agent 编排器
  class AgentOrchestrator {
    final Map<String, DivinationAgent> _agents;
    final Map<String, DivinationSkillInterface> _skills;

    /// 注册 Agent
    void registerAgent(DivinationAgent agent);

    /// 主 Agent 发起跨技法调用
    Future<AgentResponse> invokeAgent(AgentInvocation invocation) async {
      final targetAgent = _agents[invocation.toAgentId];
      final skill = _skills[targetAgent.skillId];

      // 1. 被调用 Agent 根据上下文起卦
      final divinationResult = await skill.divine(
        dateTime: invocation.context['dateTime'],
        question: invocation.context['question'],
      );

      // 2. 被调用 Agent 进行解读
      final interpretation = await _runAgentChat(
        agent: targetAgent,
        context: {
          ...invocation.context,
          'divination_result': divinationResult,
          'instruction': invocation.instruction,
        },
      );

      return AgentResponse(
        agentId: targetAgent.id,
        skillId: targetAgent.skillId,
        divinationData: divinationResult,
        interpretation: interpretation,
      );
    }
  }

  AI Chain 调用时的默认选择逻辑：

  class AgentSelector {
    /// 为指定技法选择最佳 AI 人设
    Future<AIPersona> selectBestPersona({
      required int skillId,
      String? divinationTypeUuid,
    }) async {
      // 1. 查询该技法下所有可用人设
      // 2. 按准确率、使用次数加权排序
      // 3. 返回最优人设

      final candidates = await _personaRepo.listBySkill(skillId);
      final stats = await _auditRepo.getStatsForPersonas(
        candidates.map((p) => p.uuid),
        skillId: skillId,
      );

      return candidates
        .map((p) => _PersonaWithScore(p, _calculateScore(stats[p.uuid])))
        .sorted((a, b) => b.score.compareTo(a.score))
        .first
        .persona;
    }

    double _calculateScore(PromptUsageStats? stats) {
      if (stats == null) return 0.5; // 新人设给个中等分

      // 准确率权重 60%，使用次数权重 20%，评分权重 20%
      return stats.accuracyRate * 0.6 +
             _normalizeUsageCount(stats.usageCount) * 0.2 +
             (stats.avgRating / 5.0) * 0.2;
    }
  }

  /// 溯源记录创建服务
  class ProvenanceService {

    /// 在每次 AI 调用时创建溯源记录
    Future<AIProvenance> createProvenance({
      required AIProviderConfig provider,
      required LLMModel model,
      required PromptTemplate systemPrompt,
      required Map<String, dynamic> runtimeParams,
      required Map<String, dynamic> inputContext,
      required String rawOutput,
      required AIResponseMetadata metadata,
    }) async {
      final provenance = AIProvenance(
        uuid: _uuid.v4(),

        // Provider 快照
        llmProviderUuid: provider.uuid,
        llmProviderSnapshot: jsonEncode({
          'name': provider.name,
          'api_base_url': provider.apiBaseUrl,
          'captured_at': DateTime.now().toIso8601String(),
        }),

        // Model 快照
        llmModelUuid: model.uuid,
        llmModelSnapshot: jsonEncode({
          'model_id': model.modelId,
          'model_version': model.modelVersion,
          'api_reported_version': metadata.modelVersion, // API 实际返回的版本
          'captured_at': DateTime.now().toIso8601String(),
        }),

        // Prompt 快照
        systemPromptUuid: systemPrompt.uuid,
        systemPromptSnapshot: systemPrompt.content,
        systemPromptVersion: systemPrompt.version,

        // 运行时参数
        runtimeParamsJson: jsonEncode(runtimeParams),

        // 输入快照
        inputContextHash: _sha256(jsonEncode(inputContext)),
        inputContextSnapshot: jsonEncode(inputContext),

        // 输出快照
        outputHash: _sha256(rawOutput),
        rawOutputSnapshot: rawOutput,

        // 元信息
        requestId: metadata.requestId,
        totalTokens: metadata.totalTokens,
        latencyMs: metadata.latencyMs,

        createdAt: DateTime.now(),
      );

      await _provenanceDao.insert(provenance);
      return provenance;
    }

    /// 验证溯源记录完整性
    Future<ProvenanceVerification> verify(String provenanceUuid) async {
      final provenance = await _provenanceDao.getByUuid(provenanceUuid);

      return ProvenanceVerification(
        inputHashValid: _sha256(provenance.inputContextSnapshot) ==
  provenance.inputContextHash,
        outputHashValid: _sha256(provenance.rawOutputSnapshot) ==
  provenance.outputHash,
        promptVersionMatch: await _checkPromptVersion(provenance),
        modelStillAvailable: await _checkModelAvailability(provenance),
      );
    }

    /// 尝试复现 AI 输出 (用于调试/审计)
    Future<ReproductionResult> attemptReproduction(String provenanceUuid) async
  {
      final provenance = await _provenanceDao.getByUuid(provenanceUuid);

      // 使用存储的快照重新调用
      final newOutput = await _aiService.call(
        provider: _parseProviderSnapshot(provenance.llmProviderSnapshot),
        model: _parseModelSnapshot(provenance.llmModelSnapshot),
        systemPrompt: provenance.systemPromptSnapshot,
        input: jsonDecode(provenance.inputContextSnapshot),
        params: jsonDecode(provenance.runtimeParamsJson),
      );

      return ReproductionResult(
        originalOutput: provenance.rawOutputSnapshot,
        newOutput: newOutput,
        similarity: _calculateSimilarity(provenance.rawOutputSnapshot,
  newOutput),
        isDeterministic: provenance.rawOutputSnapshot == newOutput,
      );
    }
  }

  /// Tool 定义基类
  @JsonSerializable()
  class AIToolDefinition {
    final String name;                // 工具名称 (唯一标识)
    final String description;         // 工具描述 (供 LLM 理解)
    final String category;            // 分类: divination, ui_control, 
  data_query, agent

    final Map<String, ToolParameter> parameters;
    final ToolReturnSchema? returns;

    final List<String> requiredPermissions; // 所需权限
    final bool isDestructive;         // 是否有副作用
    final int? rateLimitPerMinute;    // 调用频率限制
  }

  @JsonSerializable()
  class ToolParameter {
    final String type;                // string, integer, boolean, array, object
    final String description;
    final bool required;
    final List<String>? enumValues;   // 可选值列表
    final dynamic defaultValue;
    final Map<String, dynamic>? validation; // 验证规则
  }

  /// Tool 注册表
  abstract class ToolRegistry {
    /// 注册工具
    void register(AIToolDefinition tool, ToolExecutor executor);

    /// 获取所有工具定义 (供 LLM API 调用)
    List<Map<String, dynamic>> getToolDefinitionsForLLM();

    /// 执行工具调用
    Future<ToolResult> execute(String toolName, Map<String, dynamic> params);

    /// 获取指定技法可用的工具
    List<AIToolDefinition> getToolsForSkill(String skillId);
  }

  /// Tool 执行器接口
  typedef ToolExecutor = Future<ToolResult> Function(
    Map<String, dynamic> params,
    ToolExecutionContext context,
  );

  /// Tool 执行上下文
  class ToolExecutionContext {
    final String chatSessionUuid;
    final String? currentPanUuid;
    final String skillId;
    final Map<String, dynamic> skillInterface; // 当前技法的接口实例
  }

  /// Tool 执行结果
  @JsonSerializable()
  class ToolResult {

    /// 构建完整的 API 调用上下文
    Future<APICallContext> buildContext(String sessionUuid) async {
      final session = await _sessionDao.get(sessionUuid);
      final messages = await _messageDao.getBySession(sessionUuid);

      // 构建消息数组
      final apiMessages = <Map<String, dynamic>>[
        // 1. System prompt (始终第一条)
        {'role': 'system', 'content': session.systemPromptSnapshot},

        // 2. 历史消息 (按顺序)
        ...messages.map((m) => _toApiMessage(m)),
      ];

      // 计算 token 并可能截断
      final truncated = await _truncateIfNeeded(apiMessages);

      return APICallContext(
        messages: truncated,
        messagesHash: _sha256(jsonEncode(truncated)),
      );
    }

    /// 处理 token 限制 - 智能截断
    Future<List<Map<String, dynamic>>> _truncateIfNeeded(
      List<Map<String, dynamic>> messages,
    ) async {
      final totalTokens = await _tokenCounter.count(messages);
      final maxTokens = _config.maxContextTokens;

      if (totalTokens <= maxTokens) return messages;

      // 保留: system prompt + 最近N条消息
      // 可选: 对早期消息做摘要压缩
      return _smartTruncate(messages, maxTokens);
    }
  }

  class PromptRepository {

    Future<void> updatePrompt(String uuid, String newContent, {String?
  changeNote}) async {
      final current = await _templateDao.get(uuid);
      final newVersion = current.currentVersion + 1;

      // 1. 创建不可变版本记录
      await _versionDao.insert(PromptVersion(
        uuid: _uuid.v4(),

        version: newVersion,
        content: newContent,
        contentHash: _sha256(newContent),

        changedBy: _currentUser,
        createdAt: DateTime.now(),
      ));

      // 2. 更新主表
      await _templateDao.update(uuid,
        currentContent: newContent,
        currentVersion: newVersion,
        updatedAt: DateTime.now(),
      );
    }
  }

  验证 Prompt 完整性

  class ProvenanceVerifier {

    Future<PromptVerificationResult> verifyPromptProvenance(
      String provenanceUuid,
    ) async {
      final provenance = await _provenanceDao.get(provenanceUuid);

      // 验证1: 自身哈希一致
      final selfHashValid =
        _sha256(provenance.systemPromptContent) ==
  provenance.systemPromptContentHash;

      // 验证2: 与版本表一致
      final version = await _versionDao.get(provenance.systemPromptVersionUuid);
      final versionMatch = version != null &&
        version.content == provenance.systemPromptContent;

      // 验证3: 版本表自身完整性
      final versionHashValid = version != null &&
        _sha256(version.content) == version.contentHash;

      return PromptVerificationResult(
        selfHashValid: selfHashValid,
        versionMatch: versionMatch,
        versionHashValid: versionHashValid,
        isFullyTrusted: selfHashValid && versionMatch && versionHashValid,
      );
    }
  }

  1. 对话生命周期

  ┌─────────────────────────────────────────────────────────────────────┐

  ├─────────────────────────────────────────────────────────────────────┤

  /// 对话持久化服务
  class ChatPersistenceService {
    final AIChatSessionsDao _sessionDao;
    final AIChatMessagesDao _messageDao;
    final AIApiCallsDao _apiCallDao;
    final ProvenanceService _provenanceService;

    // ═══════════════════════════════════════════════════════════
    // 会话管理
    // ═══════════════════════════════════════════════════════════

    /// 创建新会话
    Future<AIChatSession> createSession({
      required String personaUuid,
      required String systemPromptContent,
      required String systemPromptVersionUuid,
      String? divinationUuid,
      String? panUuid,
      String? skillId,
    }) async {
      final session = AIChatSession(
        uuid: _uuid.v4(),
        personaUuid: personaUuid,
        divinationUuid: divinationUuid,
        panUuid: panUuid,
        skillId: skillId,
        systemPromptSnapshot: systemPromptContent,
        systemPromptVersionUuid: systemPromptVersionUuid,
        messageCount: 0,
        totalTokensUsed: 0,
        createdAt: DateTime.now(),
        lastActiveAt: DateTime.now(),
      );

      await _sessionDao.insert(session);
      return session;
    }

    /// 恢复会话 (加载完整历史)
    Future<ChatSessionWithHistory> loadSession(String sessionUuid) async {
      final session = await _sessionDao.get(sessionUuid);
      final messages = await _messageDao.getBySession(
        sessionUuid,
        orderBy: 'sequence_number ASC',
      );

      return ChatSessionWithHistory(
        session: session,
        messages: messages,
      );
    }

    // ═══════════════════════════════════════════════════════════
    // 消息保存
    // ═══════════════════════════════════════════════════════════

    /// 保存用户消息 (立即持久化)
    Future<AIChatMessage> saveUserMessage({
      required String sessionUuid,
      required String content,
    }) async {
      final nextSeq = await _getNextSequenceNumber(sessionUuid);

      final message = AIChatMessage(
        uuid: _uuid.v4(),
        sessionUuid: sessionUuid,
        sequenceNumber: nextSeq,
        role: 'user',
        content: content,
        createdAt: DateTime.now(),
      );

      await _messageDao.insert(message);
      await _updateSessionStats(sessionUuid);

      return message;
    }

    /// 保存 AI 响应 (流式完成后调用)
    Future<AIChatMessage> saveAssistantMessage({
      required String sessionUuid,
      required String content,
      String? toolCallsJson,
      required String apiCallUuid,
    }) async {
      final nextSeq = await _getNextSequenceNumber(sessionUuid);

      final message = AIChatMessage(
        uuid: _uuid.v4(),
        sessionUuid: sessionUuid,
        sequenceNumber: nextSeq,
        role: 'assistant',
        content: content,
        toolCallsJson: toolCallsJson,
        apiCallUuid: apiCallUuid,
        createdAt: DateTime.now(),
      );

      await _messageDao.insert(message);
      await _updateSessionStats(sessionUuid);

      return message;
    }

    /// 保存 Tool 结果
    Future<AIChatMessage> saveToolMessage({
      required String sessionUuid,
      required String toolName,
      required String toolCallId,
      required Map<String, dynamic> result,
    }) async {
      final nextSeq = await _getNextSequenceNumber(sessionUuid);

      final message = AIChatMessage(
        uuid: _uuid.v4(),
        sessionUuid: sessionUuid,
        sequenceNumber: nextSeq,
        role: 'tool',
        content: jsonEncode(result),
        toolResultsJson: jsonEncode({
          'tool_call_id': toolCallId,
          'tool_name': toolName,
          'result': result,
        }),
        createdAt: DateTime.now(),
      );

      await _messageDao.insert(message);
      return message;
    }

    // ═══════════════════════════════════════════════════════════
    // API 调用记录
    // ═══════════════════════════════════════════════════════════

    /// 记录 API 调用 (在调用前创建，调用后更新)
    Future<AIApiCall> recordApiCall({
      required String sessionUuid,
      required List<Map<String, dynamic>> messages,
      required AIProvenance provenance,
    }) async {
      final messagesJson = jsonEncode(messages);

      final apiCall = AIApiCall(
        uuid: _uuid.v4(),
        sessionUuid: sessionUuid,
        messagesSnapshot: messagesJson,
        messagesHash: _sha256(messagesJson),
        provenanceUuid: provenance.uuid,
        calledAt: DateTime.now(),
      );

      await _apiCallDao.insert(apiCall);
      return apiCall;
    }

    /// 更新 API 调用结果
    Future<void> updateApiCallResult({
      required String apiCallUuid,
      required String responseContent,
      String? responseToolCalls,
      required int inputTokens,
      required int outputTokens,
      required int latencyMs,
      String? requestId,
    }) async {
      await _apiCallDao.update(apiCallUuid,
        responseContent: responseContent,
        responseToolCalls: responseToolCalls,
        inputTokens: inputTokens,
        outputTokens: outputTokens,
        latencyMs: latencyMs,
        requestId: requestId,
      );

      // 更新 session 统计
      final apiCall = await _apiCallDao.get(apiCallUuid);
      await _sessionDao.incrementTokens(
        apiCall.sessionUuid,
        inputTokens + outputTokens,
      );
    }
  }

  /// Chat Window ViewModel
  class AIChatViewModel extends ChangeNotifier {
    final ChatPersistenceService _persistence;
    final AIProviderRepository _aiProvider;
    final ToolRegistry _toolRegistry;
    final ProvenanceService _provenanceService;

    AIChatSession? _session;
    List<ChatMessageUI> _messages = [];
    bool _isLoading = false;
    String? _streamingContent;  // 流式响应的临时内容

    List<ChatMessageUI> get messages => _messages;
    bool get isLoading => _isLoading;
    String? get streamingContent => _streamingContent;

    // ═══════════════════════════════════════════════════════════
    // 初始化
    // ═══════════════════════════════════════════════════════════

    /// 打开新对话
    Future<void> startNewSession({
      required AIPersona persona,
      String? divinationUuid,
      String? panUuid,
    }) async {
      // 获取 prompt 版本
      final promptVersion = await _promptRepo.getCurrentVersion(
        persona.systemPromptUuid,
      );

      _session = await _persistence.createSession(
        personaUuid: persona.uuid,
        systemPromptContent: promptVersion.content,
        systemPromptVersionUuid: promptVersion.uuid,
        divinationUuid: divinationUuid,
        panUuid: panUuid,
        skillId: persona.skillBindings.first,
      );

      _messages = [];
      notifyListeners();
    }

    /// 恢复历史对话
    Future<void> resumeSession(String sessionUuid) async {
      final history = await _persistence.loadSession(sessionUuid);

      _session = history.session;
      _messages = history.messages.map(_toUIMessage).toList();

      notifyListeners();
    }

    // ═══════════════════════════════════════════════════════════
    // 发送消息
    // ═══════════════════════════════════════════════════════════

    Future<void> sendMessage(String content) async {
      if (_session == null || content.trim().isEmpty) return;

      _isLoading = true;
      notifyListeners();

      try {
        // 1️⃣ 立即保存并显示用户消息
        final userMessage = await _persistence.saveUserMessage(
          sessionUuid: _session!.uuid,
          content: content,
        );
        _messages.add(_toUIMessage(userMessage));
        notifyListeners();

        // 2️⃣ 构建 API 调用上下文
        final apiMessages = _buildApiMessages();

        // 3️⃣ 创建溯源记录
        final provenance = await _provenanceService.createProvenance(
          // ... 参数
        );

        // 4️⃣ 记录 API 调用
        final apiCall = await _persistence.recordApiCall(
          sessionUuid: _session!.uuid,
          messages: apiMessages,
          provenance: provenance,
        );

        // 5️⃣ 调用 AI (流式)
        await _streamAIResponse(apiMessages, apiCall);

      } catch (e) {
        _handleError(e);
      } finally {
        _isLoading = false;
        _streamingContent = null;
        notifyListeners();
      }
    }

    /// 流式接收 AI 响应
    Future<void> _streamAIResponse(
      List<Map<String, dynamic>> messages,
      AIApiCall apiCall,
    ) async {
      final buffer = StringBuffer();
      List<ToolCall>? pendingToolCalls;
      int inputTokens = 0;
      int outputTokens = 0;
      final stopwatch = Stopwatch()..start();

      await for (final chunk in _aiProvider.chatStream(
        config: _currentConfig,
        messages: messages,
        tools: _toolRegistry.getToolDefinitionsForLLM(),
      )) {
        // 处理文本增量
        if (chunk.contentDelta != null) {
          buffer.write(chunk.contentDelta);
          _streamingContent = buffer.toString();
          notifyListeners();  // 实时更新 UI
        }

        // 处理 tool calls
        if (chunk.toolCall != null) {
          pendingToolCalls ??= [];
          pendingToolCalls.add(chunk.toolCall!);
        }

        // 处理 token 统计
        if (chunk.usage != null) {
          inputTokens = chunk.usage!.promptTokens;
          outputTokens = chunk.usage!.completionTokens;
        }
      }

      stopwatch.stop();

      // 6️⃣ 流结束，保存完整 assistant 消息
      final assistantContent = buffer.toString();
      final assistantMessage = await _persistence.saveAssistantMessage(
        sessionUuid: _session!.uuid,
        content: assistantContent,
        toolCallsJson: pendingToolCalls != null
          ? jsonEncode(pendingToolCalls.map((t) => t.toJson()).toList())
          : null,
        apiCallUuid: apiCall.uuid,
      );

      _messages.add(_toUIMessage(assistantMessage));
      _streamingContent = null;
      notifyListeners();

      // 7️⃣ 更新 API 调用结果
      await _persistence.updateApiCallResult(
        apiCallUuid: apiCall.uuid,
        responseContent: assistantContent,
        responseToolCalls: assistantMessage.toolCallsJson,
        inputTokens: inputTokens,
        outputTokens: outputTokens,
        latencyMs: stopwatch.elapsedMilliseconds,
      );

      // 8️⃣ 处理 tool calls (如果有)
      if (pendingToolCalls != null && pendingToolCalls.isNotEmpty) {
        await _handleToolCalls(pendingToolCalls);
      }
    }

    /// 处理 Tool 调用
    Future<void> _handleToolCalls(List<ToolCall> toolCalls) async {
      for (final toolCall in toolCalls) {
        // 执行工具
        final result = await _toolRegistry.execute(
          toolCall.name,
          toolCall.arguments,
        );

        // 保存 tool 消息
        final toolMessage = await _persistence.saveToolMessage(
          sessionUuid: _session!.uuid,
          toolName: toolCall.name,
          toolCallId: toolCall.id,
          result: result.toJson(),
        );

        _messages.add(_toUIMessage(toolMessage));
        notifyListeners();
      }

      // 继续 AI 响应 (带上 tool 结果)
      final apiMessages = _buildApiMessages();
      final provenance = await _provenanceService.createProvenance(/*...*/);
      final apiCall = await _persistence.recordApiCall(
        sessionUuid: _session!.uuid,
        messages: apiMessages,
        provenance: provenance,
      );

      await _streamAIResponse(apiMessages, apiCall);
    }

    /// 构建 API 消息数组
    List<Map<String, dynamic>> _buildApiMessages() {
      return [
        {'role': 'system', 'content': _session!.systemPromptSnapshot},
        ..._messages.map((m) => m.toApiMessage()),
      ];
    }
  }

  /// 对话消息 DAO
  class AIChatMessagesDao extends DatabaseAccessor<AIDatabase> {

    /// 获取会话的所有消息 (按序号排序)
    Future<List<AIChatMessage>> getBySession(
      String sessionUuid, {
      int? limit,
      int? offset,
    }) {
      return (select(aiChatMessages)
        ..where((m) => m.sessionUuid.equals(sessionUuid))
        ..orderBy([(m) => OrderingTerm.asc(m.sequenceNumber)])
        ..limit(limit ?? 1000, offset: offset)
      ).get();
    }

    /// 获取最后 N 条消息 (用于上下文截断)
    Future<List<AIChatMessage>> getLastN(String sessionUuid, int n) {
      return (select(aiChatMessages)
        ..where((m) => m.sessionUuid.equals(sessionUuid))
        ..orderBy([(m) => OrderingTerm.desc(m.sequenceNumber)])
        ..limit(n)
      ).get().then((list) => list.reversed.toList());
    }

    /// 获取下一个序号
    Future<int> getNextSequenceNumber(String sessionUuid) async {
      final result = await customSelect(
        'SELECT MAX(sequence_number) as max_seq FROM t_ai_chat_messages WHERE 
  session_uuid = ?',
        variables: [Variable.withString(sessionUuid)],
      ).getSingleOrNull();

      return (result?.read<int?>('max_seq') ?? 0) + 1;
    }
  }

  /// 会话摘要 DAO  
  class AIChatSessionsDao extends DatabaseAccessor<AIDatabase> {

    /// 获取会话列表 (带摘要信息)
    Future<List<AIChatSessionSummary>> listWithSummary({
      String? personaUuid,
      String? divinationUuid,
      int limit = 20,
      int offset = 0,
    }) async {
      final query = '''
        SELECT 
          s.*,
          p.name as persona_name,
          p.avatar as persona_avatar,
          (SELECT content FROM t_ai_chat_messages 
           WHERE session_uuid = s.uuid 
           ORDER BY sequence_number DESC LIMIT 1) as last_message_preview
        FROM t_ai_chat_sessions s
        LEFT JOIN t_ai_personas p ON s.persona_uuid = p.uuid
        WHERE s.deleted_at IS NULL
          ${personaUuid != null ? 'AND s.persona_uuid = ?' : ''}
          ${divinationUuid != null ? 'AND s.divination_uuid = ?' : ''}
        ORDER BY s.last_active_at DESC
        LIMIT ? OFFSET ?
      ''';

      final variables = <Variable>[
        if (personaUuid != null) Variable.withString(personaUuid),
        if (divinationUuid != null) Variable.withString(divinationUuid),
        Variable.withInt(limit),
        Variable.withInt(offset),
      ];

      return customSelect(query, variables: variables)
        .map(_toSessionSummary)
        .get();
    }
  }

  /// 对话导出服务
  class ChatExportService {

    /// 导出完整对话 (含溯源信息)
    Future<ChatExportData> exportSession(String sessionUuid) async {
      final session = await _sessionDao.get(sessionUuid);
      final messages = await _messageDao.getBySession(sessionUuid);
      final apiCalls = await _apiCallDao.getBySession(sessionUuid);
      final provenances = await _provenanceDao.getByIds(
        apiCalls.map((c) => c.provenanceUuid).toList(),
      );

      return ChatExportData(
        exportVersion: '1.0',
        exportedAt: DateTime.now(),
        session: session.toJson(),
        messages: messages.map((m) => m.toJson()).toList(),
        apiCalls: apiCalls.map((c) => c.toJson()).toList(),
        provenances: provenances.map((p) => p.toJson()).toList(),
      );
    }

    /// 导出为 Markdown (人类可读)
    Future<String> exportAsMarkdown(String sessionUuid) async {
      final session = await _sessionDao.get(sessionUuid);
      final messages = await _messageDao.getBySession(sessionUuid);
      final persona = await _personaDao.get(session.personaUuid);

      final buffer = StringBuffer();
      buffer.writeln('# AI 对话记录');
      buffer.writeln();
      buffer.writeln('**AI 人设**: ${persona.name} (${persona.title})');
      buffer.writeln('**创建时间**: ${session.createdAt}');
      buffer.writeln('**消息数量**: ${session.messageCount}');
      buffer.writeln();
      buffer.writeln('---');
      buffer.writeln();

      for (final message in messages) {
        final roleLabel = switch (message.role) {
          'user' => '👤 用户',
          'assistant' => '🤖 ${persona.name}',
          'tool' => '🔧 工具调用',
          _ => message.role,
        };

        buffer.writeln('### $roleLabel');
        buffer.writeln();
        buffer.writeln(message.content ?? '');
        buffer.writeln();
      }

      return buffer.toString();
    }
  }
